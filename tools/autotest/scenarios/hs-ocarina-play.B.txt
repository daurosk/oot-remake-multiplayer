# load) and tries again. Notes as in ocarina-probe: A = A, L = Down, R = Right, Y = Left, X = Up.

wait 120

wait 40
press X 12
wait 40
press R 12
wait 40
press Y 12
wait 40
press X 12
wait 40
press R 12
wait 300
mark lullaby
press B 12
wait 120
press B 12
wait 120
stick 0 -1 120          # back down the path, out of the Graveyard (as hs-items-share)

wait 300
press DLEFT 12
wait 240
mark out-2

wait 120
mark note-a-2

press B 12
