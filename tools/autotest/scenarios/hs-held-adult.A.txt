# hs-held-adult (Hookshot): held-adult for Hookshot/Longshot. The lane saves are child (Graveyard), and
# Hookshot's warp is the bridge's teleport, which keeps the save's age (held-adult on lane 1, 2026-11-16:
# "sent (map KakarikoVillage_Child)"), so both games set Sequence_IsChildhood 0 first, as hs-adult does.
# run.ps1 loads the save (no noload). A is given the bow, the hookshot, the hammer and bombs, steps aside
# to face B and uses each item on ZR (assign puts it there).
# Run: tools\autotest\run.ps1 -Runtime hookshot -Lane 1 -Scenario hs-held-adult
# Pass, in B's log: "P1 wears held 'Weapon_Range_Hookshot" (#232/#233: the Hookshot on the puppet) and no
# "CRASH:" or "a write: it faults next" in either log; "P1 wears held '<item>_Blade1'" for the hammer
# (a wearer copy of its own). Shots item1-5: the bow, hookshot, hammer, bomb, bow in the puppet's hand.
#! switches LICHII_COOP_PUPPET=1,LICHII_COOP_CHEATS=1,LICHII_COOP_OBJECTS_CHEAT=1,LICHII_COOP_DEBUG_ANIM=1
mark start
setflag Sequence_IsChildhood 0
warp KakarikoVillage_Adult
wait 900                # the warp's load, the puppets made
mark after
give bow
give hookshot
give PouchData.Item_Hammer.StockNum
give bombs
wait 600
stick 0 0.6 100
stick 0.7 0 30
stick 0 -0.4 12
wait 60
mark before
assign Bow
wait 30
hold ZR 90 item1       # the mark halfway, the item still out
wait 30
press B
wait 90
assign Hookshot
wait 30
hold ZR 90 item2
wait 30
press B
wait 90
assign Hammer
wait 30
hold ZR 90 item3
wait 30
press B
wait 90
assign Bomb
wait 30
hold ZR 90 item4
wait 30
press B
wait 90
assign Bow
wait 30
hold ZR 90 item5
wait 30
press B
wait 150
mark end
