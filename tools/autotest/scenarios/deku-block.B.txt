# deku-block, B: put in the basement beside A, watches A's push, then pushes the same block
# twice from where A left it (see deku-block.A.txt).
mark start
warp InsideTheDekuTree Entrance
wait 1200
mark title
press A
wait 180
press A
wait 180
press A
wait 300
press A
wait 2400
mark after
place -54.5 79.9 50.2    # beside A, out of its way
wait 240
toward -51.06 47.0 10
press ZL 10
mark b-placed
wait 600                 # A pushes now
mark b-watched
place -51.06 79.9 50.2   # behind the block, where A stood
wait 240
goto -51.06 47.4 0.3 300 # walk up to its +z face (block one step on, at about 45.6)
mark b-reach
press A 90               # GRAB by the non-owner, held 1.5 s: the block must not move
mark b-grab
toward -51.06 60.0 120 A # PULL it back one step toward +z, A held
wait 120
mark b-pull
toward -51.06 40.0 150 A # PUSH it on, A held (grab and push)
wait 120
mark b-push1
toward -51.06 40.0 150   # and once more, walking into it
wait 120
mark b-push2
wait 1200
mark b-end
