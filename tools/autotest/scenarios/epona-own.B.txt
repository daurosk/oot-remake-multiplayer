# epona-own, B: the first rider (see epona-own.A.txt for the run and the pass lines).
# Rides its own Epona, gets off, then says Epona is won (the quest Get_Epona's Enum 0x92431c55 to
# Complete, as winning the race does; shared by its end, lichii/hook/coop_quests.cpp) and rides
# again while A rides its own.
#! switches LICHII_COOP_DEBUG_ANIM=1
mark start
setflag Sequence_IsChildhood 0  # adult (Hookshot: run.ps1 loads the save first)
warp HyruleField_Adult HyruleField_FrontOfMarket
wait 1200               # the warp's load
mark field
press PLUS 12           # probe: the pause menu must open (scripted buttons reach the game)
wait 60
mark probe-pause
press PLUS 12           # close it
wait 120
place -95.5 151.0 264.5 # 5 m east of Epona
wait 240
goto -99.3 264.55 0.5 180 # walk to her side: Link ends facing her (toward's one push faced him away)
wait 30
mark by-horse
press A 12              # get on
wait 40
mark mounted            # (an unowned Epona throws Link off after about 1.5 s)
stick 0 0.8 120         # ride forward
mark riding
stick 0 0.8 120
mark riding2
wait 240                # let her stop
press A 12              # get off
wait 240
mark dismounted
wait 300
mark owned              # (name kept: same time as A's)
wait 270                # she stands where we got off
press A 12              # get on again
wait 240
mark both-mounted
stick 0 0.8 240         # ride forward while A rides too
mark both-riding
wait 120
press A 12              # get off
wait 240
mark end
setenum 0x92431c55 Complete   # Epona won (Get_Epona): only now, the field Epona leaves both games
wait 300
mark won
