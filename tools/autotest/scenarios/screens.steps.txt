# screens (issue 86): the title screen, file select and the pause menu, each both ways. Counts are puppets
# (CoopPuppet_*) and their Navis in each game, from the objects debug log.
# Run: tools\autotest\run.ps1 -Scenario screens
start A B
load A                         # A plays; B stays on the title screen
expect b-on-title A=0 B=0      # B sends no samples: no puppet of B in A; none in B's title
load B
expect both-in A=1 B=1
play B pause.B.txt             # B opens the pause menu for 6 s and closes it
expect b-paused-back A=1 B=1   # A kept B's puppet standing; B's game kept A's
play A pause.A.txt             # the same the other way, A walking first so B sees it stop
expect a-paused-back A=1 B=1
close A                        # A quits; it comes back on the title screen while B plays
start A
expect a-on-title A=0 B=0      # the other way: A on the title, so no puppet of A in B; none in A
load A
expect a-back-in A=1 B=1

# menu (whether the puppet is drawn behind it is the open question); the other window shows
# the paused player's puppet standing still, not gone.
