# screens (issue 86): the title screen, file select and the pause menu, each both ways. Counts are puppets
# (CoopPuppet_*) and their Navis in each game, from the objects debug log.
# Run: tools\autotest\run.ps1 -Scenario screens
start A B
load A                         # A plays; B stays on the title screen
expect b-on-title A=0 B=0  see=screens-b-on-title-A  see=screens-b-on-title-B      # B sends no samples: no puppet of B in A; none in B's title
load B
expect both-in A=1 B=1  see=screens-both-in-A  see=screens-both-in-B
play B pause.B.txt             # B opens the pause menu for 6 s and closes it
expect b-paused-back A=1 B=1  see=screens-b-paused-back-A  see=screens-b-paused-back-B   # A kept B's puppet standing; B's game kept A's
play A pause.A.txt             # the same the other way, A walking first so B sees it stop
expect a-paused-back A=1 B=1  see=screens-a-paused-back-A  see=screens-a-paused-back-B
close A                        # A quits; it comes back on the title screen while B plays
start A
expect a-on-title A=0 B=0  see=screens-a-on-title-A  see=screens-a-on-title-B      # the other way: A on the title, so no puppet of A in B; none in A
# Not covered: going back to the title FROM play (quit in the menu: fade 0.4 s, deleted after the
# 2 s grace) while still connected; close A leaves the server. By hand or a later scenario.
load A
expect a-back-in A=1 B=1  see=screens-a-back-in-A  see=screens-a-back-in-B
